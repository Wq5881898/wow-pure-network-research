cmake_minimum_required(VERSION 3.20)
project(AgentWowNav LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(DETOUR_ROOT "" CACHE PATH "Pinned local AzerothCore deps/recastnavigation/Detour")
if(NOT EXISTS "${DETOUR_ROOT}/Include/DetourNavMesh.h")
  message(FATAL_ERROR "Set DETOUR_ROOT to the AzerothCore Detour source directory")
endif()
file(GLOB DETOUR_SOURCES CONFIGURE_DEPENDS "${DETOUR_ROOT}/Source/*.cpp")
add_executable(agent_wow_nav nav.cpp ${DETOUR_SOURCES})
target_include_directories(agent_wow_nav PRIVATE "${DETOUR_ROOT}/Include")
